CREATE TABLE users (
  id            BIGINT AUTO_INCREMENT PRIMARY KEY,
  username      VARCHAR(50)  NOT NULL UNIQUE,   -- 로그인 ID
  email         VARCHAR(100) NOT NULL UNIQUE,   -- career-hi 호환
  password_hash VARCHAR(255) NOT NULL,
  role          VARCHAR(20)  NOT NULL DEFAULT 'USER',  -- ADMIN/USER
  -- univNotice용 추가 컬럼 (null 허용 — 서비스 미사용 시 null)
  school_id     INT NULL,
  department_id INT NULL,
  phone_hash    VARCHAR(255) NULL,              -- univNotice 로그인용
  student_id_hash VARCHAR(255) NULL,
  marketing_agreed BOOLEAN DEFAULT FALSE,       -- career-hi 호환
  created_at    DATETIME NOT NULL DEFAULT NOW(),
  updated_at    DATETIME NOT NULL DEFAULT NOW() ON UPDATE NOW()
);
⚠️ univNotice는 학번+폰번호 기반 인증 구조로, 표준 username/password 로그인과 완전히 다릅니다. 포트폴리오 데모 환경에서는 공통 계정 1개만 사용하므로, univNotice 전용 인증 로직을 common에 통합할 필요 없이 해당 필드만 USERS에 추가해두면 됩니다.

ganzithon 최적화
① JWT 필터 — 매 요청마다 DB 조회 제거
문제: JwtAuthenticationFilter가 UserRepository.findById()를 매 요청마다 실행
개선: JWT 클레임에 userId + email + role을 담아 DB 조회 없이 인증 처리
효과: 인증 처리당 SELECT 1회 제거 → 초당 100 요청 기준 100 QPS 감소
DB 조회 감소율: ~100% (인증 요청 기준)
② ReportController — @AuthenticationPrincipal 타입 불일치 수정
문제: @AuthenticationPrincipal User user — ganzithon 전용 User 엔티티에 의존
개선: @AuthenticationPrincipal UserDetails userDetails로 변경, email로 userId 조회 분리
효과: common User 엔티티로 교체 후에도 동일하게 동작
③ ReportApi 인터페이스 — 불필요한 추상화 제거
문제: ReportApi 인터페이스 → ReportController 구현 구조가 과도한 추상화
개선: 단일 모듈에서 인터페이스 분리는 불필요. Controller에 직접 Swagger 어노테이션 통합
효과: 코드 경로 단순화, IDE 네비게이션 개선
career-hi 최적화
① 테이블명 충돌 수정 — @Table(name) 명시
문제: Report.java의 @Table이 없거나 "reports" — ganzithon과 충돌
개선: @Table(name = "ch_report") 명시적 지정
효과: 통합 DB에서 테이블 충돌 방지
② Profile 스펙 — @ElementCollection → 별도 테이블로 정규화
❌ 기존: @ElementCollection 4개 컬렉션
✅ 개선: CH_PROFILE_SPEC 단일 테이블
// 4개의 별도 조인 테이블 생성됨 @ElementCollection List<String> certificates; @ElementCollection List<String> codingLanguages; @ElementCollection List<LanguageTest> languageTests; @ElementCollection List<Award> awards; // 조회 시 4번의 SELECT 발생

프로필 조회 쿼리 수: 5회 → 1회 (80% 감소)
③ AI 분석 결과 저장 — aiAnalysisJson(TEXT) 구조 개선

univNotice 최적화
① 인증번호 조회 — find_phone 후보군 방식 최적화 유지
기존 방식 (잘 설계됨): 번호 뒷 4자리(find_phone)로 후보를 좁힌 후 bcrypt 비교
이유: 폰번호를 bcrypt로 해시했기 때문에 direct WHERE 불가능 — 뒷 4자리 인덱스로 O(n) → O(1) 근사
통합 시 유지: 이 패턴은 common의 PhoneVerification 테이블에 그대로 적용
✅ univNotice의 find_phone / find_student_id 패턴은 암호화된 개인정보를 효율적으로 검색하는 좋은 설계입니다. 통합 후에도 동일하게 유지합니다.
② 공지 조회 — N+1 문제 해결
문제: CrawlUrl.findAll에서 include: [Notice, Category, Keyword] 3중 중첩 — separate: true로 N+1을 일부 해결했지만 여전히 쿼리 3~4회 발생
개선: Spring Boot JPA 전환 시 @EntityGraph로 한 번의 LEFT JOIN FETCH로 처리
효과: 공지 목록 조회 쿼리 수 3~4회 → 1회
쿼리 수 감소: ~75%
③ 인증번호 만료 레코드 — 자동 정리
문제: PhoneVerification 테이블에 만료된 레코드가 계속 누적됨
개선: Spring Scheduler로 매 1시간마다 DELETE FROM verification_codes WHERE expires_at < NOW() 실행
효과: 테이블 크기 제어, find_phone 후보군 조회 성능 유지
ESC 최적화
① email FK → user_id(bigint) FK 통일
문제: ShoppingBag, Payment, CancelPayment 등이 userEmail VARCHAR를 FK로 사용
개선: 모든 FK를 user_id BIGINT로 변경
효과: VARCHAR 비교 → BIGINT 비교 (인덱스 탐색 ~3배 빠름), JOIN 비용 감소
인덱스 탐색 효율: VARCHAR(50) 대비 BIGINT PK 기준 약 3배 향상
② 상품 조회 — 복수 Include 최적화
문제: Product.findOne에 include 6개 (Tag, Discount, Img, Option, OptionProperty, Review) — 다중 LEFT JOIN + N+1

③ 결제 조회 — purchaseProductInformation(JSON) 파싱 제거
문제: Payment.purchaseProductInformation을 JSON으로 저장 후 JS에서 파싱해 productId 추출 → 별도 Product 조회
개선: ESC_PAYMENT_ITEM 테이블을 별도로 두어 정규화 (payment_id, product_id, quantity, options)
효과: JSON 파싱 제거, 상품별 주문 내역 쿼리 가능, 인덱스 활용 가능
④ grade(별점) 업데이트 — 매 조회마다 재계산 제거
문제: 상품 상세 조회 시 모든 리뷰의 평균을 계산하고 grade 불일치 시 UPDATE 실행
